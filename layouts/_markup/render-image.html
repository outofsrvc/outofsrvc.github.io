{{- /* Image render hook. Resolves page/asset resources when the destination names one, and otherwise leaves the path untouched. This site is served from the domain root, so site-relative paths such as /assets/img/workshop/... already resolve in both `hugo server` and production — no BaseURL rewriting needed (unlike the standalone workshop, which lived under a project subpath and had to reconstruct them). Adds lazy loading and async decoding to every content image. */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $src = .RelPermalink -}} {{- with $u.RawQuery -}} {{- $src = printf "%s?%s" $src . -}} {{- end -}} {{- with $u.Fragment -}} {{- $src = printf "%s#%s" $src . -}} {{- end -}} {{- end -}} {{- end -}} {{- $attributes := merge .Attributes (dict "alt" .Text "src" $src "title" (.Title | transform.HTMLEscape) "loading" "lazy" "decoding" "async") -}} {{- /**/ -}}
